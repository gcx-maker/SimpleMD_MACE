cmake_minimum_required(VERSION 3.18)

project(simplemd_pybind)


# =========================
# C++ standard
# =========================

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)



# =========================
# Python
# =========================

find_package(Python COMPONENTS Interpreter Development REQUIRED)


# =========================
# pybind11
# =========================

find_package(pybind11 CONFIG REQUIRED)



# =========================
# Source files
# =========================

add_executable(main

	main.cpp
	
    input.cpp
	integrator.cpp
	
    system.cpp
    Vector.cpp
    Random.cpp

    xyzreader.cpp

    neighbor.cpp


    thermostat.cpp

    output.cpp
    setup.cpp


    ljforcefield.cpp
    maceforcefield.cpp

)


# =========================
# Include
# =========================

target_include_directories(main PRIVATE

    ${Python_INCLUDE_DIRS}

)



# =========================
# Link
# =========================

target_link_libraries(main PRIVATE

    ${Python_LIBRARIES}

    pybind11::embed

)
